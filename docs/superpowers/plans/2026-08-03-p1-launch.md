# P1 Launch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the funded first build (P1) of the folk-medicine web app: a single self-hosted binary that lets district staff enter healer/recipe/case data and lets the public read and search it.

**Architecture:** One Go binary serves a JSON API and an embedded Next.js static export (ADR-0001). Data is in SQLite through a portable data-access layer (standard SQL only, so a later move to PostgreSQL is cheap). The binary terminates TLS itself with ACME. The Next.js app is a client-rendered SPA that calls the JSON API with a session cookie; Node is a build-time tool only.

**Tech Stack:** Go (stdlib `net/http` `ServeMux`, Go 1.22+ method routing), `modernc.org/sqlite` (pure Go, cgo-free), `golang.org/x/crypto/bcrypt`, `golang.org/x/crypto/acme/autocert`, `github.com/disintegration/imaging`, `github.com/xuri/excelize/v2`; Next.js 15 / React 19 static export, plain `fetch`.

## Global Constraints

- **TDD, always.** Failing test → confirm fail → minimal code → confirm pass → refactor. No exceptions.
- **Single binary, cgo-free.** Use `modernc.org/sqlite` (driver name `sqlite`). Never introduce a cgo dependency; the binary must cross-compile and run as one static file.
- **Portable SQL only** (ADR-0001). No SQLite-only syntax. Placeholders are `?`; timestamps are TEXT `RFC3339`; booleans are INTEGER `0/1`. All DB access goes through `internal/db`.
- **Go style:** uber-go style guide; modern Go idioms; organized imports (stdlib, third-party, local — sorted).
- **JS/TS style:** Google TS/JS style guide; organized imports.
- **PDPA:** private fields (`phone`, `consent_obtained`, `consent_date`, `updated_by`, `updated_at`) are never in any public response or payload.
- **i18n:** Thai is the UI default; an English toggle switches labels only; record content is never translated.
- **Commands:** prefix shell commands with `rtk` (e.g. `rtk go test ./...`).
- **Commit** after every green step. Small commits.
- **Scale targets (provisional):** ~500 doctors, ~5k recipes; list < 1 s, search < 2 s.
- **Reference:** the SRS (`docs/superpowers/specs/2026-08-03-srs.md`) and the data model (`docs/superpowers/specs/2026-08-03-data-model-and-form-design.md`). Requirement IDs below (FR-*, NFR-*) point to the SRS.

package httpx

import (
	"encoding/json"
	"net/http"
)

// WriteJSON writes v as JSON with the given status.
func WriteJSON(w http.ResponseWriter, status int, v any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(v)
}

// WriteError writes a JSON error body {"error":{"code","message"}}.
func WriteError(w http.ResponseWriter, status int, code, msg string) {
	WriteJSON(w, status, map[string]any{
		"error": map[string]string{"code": code, "message": msg},
	})
}

// DecodeJSON decodes the request body into v, rejecting unknown fields.
func DecodeJSON(r *http.Request, v any) error {
	dec := json.NewDecoder(r.Body)
	dec.DisallowUnknownFields()
	return dec.Decode(v)
}
```

package db

import (
	"database/sql"
	"fmt"
	"strings"

	_ "modernc.org/sqlite"
)

// DB wraps *sql.DB and records the driver so SQL can be rebound per dialect.
type DB struct {
	*sql.DB
	Driver string
}

// Open opens a SQLite database at path with sane pragmas for a small server.
func Open(path string) (*DB, error) {
	dsn := fmt.Sprintf("file:%s?_pragma=busy_timeout(5000)&_pragma=journal_mode(WAL)&_pragma=foreign_keys(1)", path)
	sqlDB, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, fmt.Errorf("open sqlite: %w", err)
	}
	sqlDB.SetMaxOpenConns(1) // WAL + single writer keeps SQLite safe under our load
	return &DB{DB: sqlDB, Driver: "sqlite"}, nil
}

// Rebind converts `?` placeholders to the driver's dialect.
// SQLite keeps `?`; Postgres (future) becomes `$1, $2, ...`.
func (d *DB) Rebind(query string) string {
	if d.Driver != "postgres" {
		return query
	}
	var b strings.Builder
	n := 0
	for _, r := range query {
		if r == '?' {
			n++
			fmt.Fprintf(&b, "$%d", n)
			continue
		}
		b.WriteRune(r)
	}
	return b.String()
}
```

package db

import (
	"embed"
	"fmt"
	"io/fs"
	"sort"
	"strings"
	"time"
)

//go:embed migrations/*.sql
var migrationFS embed.FS

// Migrate applies each embedded migration once, in filename order.
func Migrate(d *DB) error {
	if _, err := d.Exec(`CREATE TABLE IF NOT EXISTS schema_migration (
		version TEXT PRIMARY KEY, applied_at TEXT NOT NULL)`); err != nil {
		return fmt.Errorf("ensure schema_migration: %w", err)
	}
	entries, err := fs.ReadDir(migrationFS, "migrations")
	if err != nil {
		return err
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		names = append(names, e.Name())
	}
	sort.Strings(names)

	for _, name := range names {
		version := strings.TrimSuffix(name, ".sql")
		var exists int
		if err := d.QueryRow(`SELECT count(*) FROM schema_migration WHERE version = ?`, version).Scan(&exists); err != nil {
			return err
		}
		if exists > 0 {
			continue
		}
		body, err := migrationFS.ReadFile("migrations/" + name)
		if err != nil {
			return err
		}
		if _, err := d.Exec(string(body)); err != nil {
			return fmt.Errorf("apply %s: %w", name, err)
		}
		if _, err := d.Exec(`INSERT INTO schema_migration (version, applied_at) VALUES (?, ?)`,
			version, time.Now().UTC().Format(time.RFC3339)); err != nil {
			return err
		}
	}
	return nil
}
```

import (
	"embed"

	"net/http"

//go:embed all:dist
var distFS embed.FS

// Handler serves the embedded Next.js export. Unknown non-API, non-asset
// paths fall back to index.html so client-side routes work.
func Handler() http.Handler {
	sub, err := fs.Sub(distFS, "dist")
	if err != nil {
		panic(err)
	}
	files := http.FileServer(http.FS(sub))
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if _, err := fs.Stat(sub, strings.TrimPrefix(r.URL.Path, "/")); err != nil && r.URL.Path != "/" {
			r = r.Clone(r.Context())
			r.URL.Path = "/"
		}
		files.ServeHTTP(w, r)
	})
}

package httpx

import (
	"context"
	"crypto/tls"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"golang.org/x/crypto/acme/autocert"

	"phum-panya/internal/config"
)

// Serve runs handler with graceful shutdown. Dev mode is plain HTTP; prod
// terminates TLS with autocert and redirects :80 → :443.
func Serve(cfg config.Config, handler http.Handler) error {
	srv := &http.Server{
		Handler:           handler,
		ReadHeaderTimeout: 10 * time.Second,
	}
	go func() {
		if cfg.DevMode || cfg.Domain == "" {
			srv.Addr = cfg.HTTPAddr
			_ = srv.ListenAndServe()
			return
		}
		m := &autocert.Manager{
			Prompt:     autocert.AcceptTOS,
			HostPolicy: autocert.HostWhitelist(cfg.Domain),
			Cache:      autocert.DirCache("data/certs"),
		}
		srv.Addr = ":443"
		srv.TLSConfig = &tls.Config{GetCertificate: m.GetCertificate, MinVersion: tls.VersionTLS12}
		go http.ListenAndServe(":80", m.HTTPHandler(nil)) // ACME + redirect
		_ = srv.ListenAndServeTLS("", "")
	}()

	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
	<-stop
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	return srv.Shutdown(ctx)
}
```

```go
package auth

import (
	"crypto/rand"
	"crypto/sha256"
	"database/sql"
	"encoding/base64"
	"encoding/hex"
	"errors"
	"net/http"
	"time"

	"phum-panya/internal/clock"
	"phum-panya/internal/db"
)

var ErrNoSession = errors.New("no valid session")

type Session struct {
	UserID    int64
	CSRF      string
	ExpiresAt time.Time
}

type SessionStore struct {
	db  *db.DB
	clk clock.Clock
	ttl time.Duration
}

func NewSessionStore(d *db.DB, clk clock.Clock, ttl time.Duration) *SessionStore {
	return &SessionStore{db: d, clk: clk, ttl: ttl}
}

func randToken() string {
	b := make([]byte, 32)
	_, _ = rand.Read(b)
	return base64.RawURLEncoding.EncodeToString(b)
}

func hashToken(t string) string {
	sum := sha256.Sum256([]byte(t))
	return hex.EncodeToString(sum[:])
}

func (s *SessionStore) Create(userID int64) (string, string, error) {
	tok := randToken()
	csrf := randToken()
	exp := s.clk.Now().Add(s.ttl).Format(time.RFC3339)
	_, err := s.db.Exec(s.db.Rebind(
		`INSERT INTO app_session(token_hash,user_id,csrf,expires_at) VALUES(?,?,?,?)`),
		hashToken(tok), userID, csrf, exp)
	if err != nil {
		return "", "", err
	}
	return tok, csrf, nil
}

func (s *SessionStore) Lookup(tok string) (*Session, error) {
	var uid int64
	var csrf, exp string
	err := s.db.QueryRow(s.db.Rebind(
		`SELECT user_id,csrf,expires_at FROM app_session WHERE token_hash=?`),
		hashToken(tok)).Scan(&uid, &csrf, &exp)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, ErrNoSession
	}
	if err != nil {
		return nil, err
	}
	t, _ := time.Parse(time.RFC3339, exp)
	if !t.After(s.clk.Now()) {
		return nil, ErrNoSession
	}
	return &Session{UserID: uid, CSRF: csrf, ExpiresAt: t}, nil
}

func (s *SessionStore) Delete(tok string) error {
	_, err := s.db.Exec(s.db.Rebind(`DELETE FROM app_session WHERE token_hash=?`), hashToken(tok))
	return err
}

func SetSessionCookie(w http.ResponseWriter, tok, csrf string, secure bool) {
	http.SetCookie(w, &http.Cookie{Name: "session", Value: tok, Path: "/", HttpOnly: true,
		Secure: secure, SameSite: http.SameSiteLaxMode})
	http.SetCookie(w, &http.Cookie{Name: "csrf", Value: csrf, Path: "/",
		Secure: secure, SameSite: http.SameSiteLaxMode})
}

func ClearSessionCookie(w http.ResponseWriter) {
	for _, n := range []string{"session", "csrf"} {
		http.SetCookie(w, &http.Cookie{Name: n, Value: "", Path: "/", MaxAge: -1})
	}
}
```
